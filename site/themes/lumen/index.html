import os, requests

r = requests.post(
    "https://api.example.com/v1/generate",
    headers={"Authorization": f"Bearer {os.environ['QUARRY_API_KEY']}"},
    json={"deployment": "support-bot", "prompt": "Where is my order?"},
)
print(r.json()["output"])
const r = await fetch("https://api.example.com/v1/generate", {
  method: "POST",
  headers: { Authorization: `Bearer ${process.env.QUARRY_API_KEY}` },
  body: JSON.stringify({ deployment: "support-bot", prompt: "Where is my order?" }),
});
console.log((await r.json()).output);
